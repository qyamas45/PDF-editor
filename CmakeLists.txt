cmake_minimum_required(VERSION 3.16)
project(PDFEditor VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# MSVC requires this flag so Qt can detect the C++17 standard correctly
if(MSVC)
    add_compile_options(/Zc:__cplusplus)
    add_compile_options(/W4 /permissive-)
endif()

# Enable Qt auto-processing
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)

# Find Qt6 with required modules
find_package(Qt6 REQUIRED COMPONENTS
    Core
    Gui
    Widgets
    Pdf        # Uncomment after installing Qt PDF via Qt Maintenance Tool
    PdfWidgets # Uncomment after installing Qt PDF via Qt Maintenance Tool
)

# Source files
set(SOURCES
    src/main.cpp
    src/ui/MainWindow.cpp
)

# Create executable
add_executable(PDFEditor ${SOURCES})

# Allow includes relative to src/ e.g. #include "core/PDFDocument.h"
target_include_directories(PDFEditor PRIVATE src)

# Link Qt6 libraries
target_link_libraries(PDFEditor PRIVATE
    Qt6::Core
    Qt6::Gui
    Qt6::Widgets
    Qt6::Pdf        # Uncomment after installing Qt PDF via Qt Maintenance Tool
    Qt6::PdfWidgets # Uncomment after installing Qt PDF via Qt Maintenance Tool
)

# Automatically deploy Qt DLLs next to the exe after every build (Windows only)
if(WIN32)
    find_program(WINDEPLOYQT windeployqt HINTS "${Qt6_DIR}/../../../bin")
    if(WINDEPLOYQT)
        add_custom_command(TARGET PDFEditor POST_BUILD
            COMMAND ${WINDEPLOYQT}
                --no-translations
                --no-system-d3d-compiler
                $<TARGET_FILE:PDFEditor>
            COMMENT "Deploying Qt DLLs..."
        )
    endif()
endif()
